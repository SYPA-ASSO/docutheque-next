// lib/diagnostic-account.ts
// Rattache une commande de diagnostic RGPD a un espace client.
// A la commande : on retrouve le compte par users.email (source de verite), sinon on le cree
// (sans mot de passe) et on prepare un lien de definition de mot de passe (recuperation Supabase)
// a envoyer par email. On enregistre ensuite la commande dans la table dossiers_rgpd.
// Tout passe par le client service-role (supabaseAdmin), qui contourne la RLS.
// Regles : zero tiret long dans le contenu visible, NBSP avant euro, Notre juriste singulier.

import { supabaseAdmin } from '@/lib/supabase-admin'

export type CompteDiagnostic = {
  userId: string | null
  created: boolean
  accessLink: string | null
}

// Separe un nom complet en prenom + nom (premier mot = prenom, reste = nom).
export function splitFullName(fullName: string): { prenom: string; nom: string } {
  const propre = (fullName || '').trim().replace(/\s+/g, ' ')
  if (!propre) return { prenom: 'Client', nom: '' }
  const parts = propre.split(' ')
  if (parts.length === 1) return { prenom: parts[0], nom: '' }
  return { prenom: parts[0], nom: parts.slice(1).join(' ') }
}

// Recherche l'identifiant Supabase Auth d'un utilisateur par email (parcours pagine, defensif).
// Utilise seulement dans le cas limite ou un compte Auth existe sans ligne users correspondante.
async function findAuthUserIdByEmail(email: string): Promise<string | null> {
  const cible = email.toLowerCase()
  for (let page = 1; page <= 5; page++) {
    const { data, error } = await supabaseAdmin.auth.admin.listUsers({ page, perPage: 200 })
    if (error || !data || !data.users || data.users.length === 0) return null
    const trouve = data.users.find(u => (u.email || '').toLowerCase() === cible)
    if (trouve) return trouve.id
    if (data.users.length < 200) return null
  }
  return null
}

// Retrouve un compte par email, sinon le cree. Retourne l'identifiant, l'etat de creation
// et, si le compte vient d'etre cree, un lien de definition de mot de passe a envoyer par email.
export async function findOrCreateDiagnosticUser(params: {
  email: string
  fullName: string
  telephone?: string | null
  structure?: string | null
  origin: string
}): Promise<CompteDiagnostic> {
  const email = (params.email || '').trim().toLowerCase()
  if (!email) return { userId: null, created: false, accessLink: null }

  // 1. Compte deja existant ? (la table users est alimentee a chaque inscription)
  const { data: existant } = await supabaseAdmin
    .from('users')
    .select('id')
    .eq('email', email)
    .maybeSingle()
  if (existant && existant.id) {
    return { userId: existant.id, created: false, accessLink: null }
  }

  // 2. Creation du compte Auth (sans mot de passe : defini ensuite via le lien de recuperation)
  let userId: string | null = null
  let created = false
  const { data: authData, error: authError } = await supabaseAdmin.auth.admin.createUser({
    email,
    email_confirm: true,
  })
  if (authError) {
    // Cas limite : compte Auth existant sans ligne users. On recupere son id et on le
    // traite comme un compte existant (aucun lien de mot de passe envoye).
    if ((authError.message || '').toLowerCase().includes('registered')) {
      userId = await findAuthUserIdByEmail(email)
      if (!userId) throw new Error('Compte Auth existant introuvable : ' + authError.message)
      created = false
    } else {
      throw authError
    }
  } else {
    userId = authData.user.id
    created = true
  }

  // 3. Profil dans la table users (upsert pour couvrir le cas limite ci-dessus)
  const { prenom, nom } = splitFullName(params.fullName)
  const { error: profilError } = await supabaseAdmin
    .from('users')
    .upsert(
      {
        id: userId,
        email,
        prenom,
        nom,
        type: 'professionnel',
        telephone: params.telephone || null,
        nom_structure: params.structure || null,
        newsletter: false,
        abonnement_actif: false,
        role: 'user',
      },
      { onConflict: 'id' }
    )
  if (profilError) throw profilError

  // 4. Lien de definition de mot de passe (recuperation) vers la page /nouveau-mot-de-passe
  let accessLink: string | null = null
  if (created) {
    try {
      const { data: linkData } = await supabaseAdmin.auth.admin.generateLink({
        type: 'recovery',
        email,
        options: { redirectTo: `${params.origin}/nouveau-mot-de-passe` },
      })
      accessLink = (linkData && linkData.properties && linkData.properties.action_link) || null
    } catch (e) {
      console.error('[diagnostic-account] generateLink recovery a echoue :', e)
    }
  }

  return { userId, created, accessLink }
}

// Verifie si une commande carte a deja ete enregistree (idempotence cote serveur).
export async function dossierExistsForSession(stripeSessionId: string): Promise<boolean> {
  const { data } = await supabaseAdmin
    .from('dossiers_rgpd')
    .select('id')
    .eq('stripe_session_id', stripeSessionId)
    .maybeSingle()
  return Boolean(data && data.id)
}

export async function createDossierRgpd(params: {
  userId: string
  forfaitId: 'essentiel' | 'standard'
  montantEuros: number
  mode: 'carte' | 'virement'
  reference?: string | null
  stripeSessionId?: string | null
  statut: 'en_attente' | 'en_cours' | 'livre'
}): Promise<void> {

// lib/tarifs.ts
// Tarifs des abonnements et calcul du revenu mensuel recurrent (MRR).
// Equivalent mensuel en euros par type et periode (annuel ramene au mois).

export const MRR_PAR_ABONNEMENT: Record<string, number> = {
  'optimal_particulier|monthly': 17.9,
  'optimal_particulier|yearly': 149 / 12,
  'optimal_pro|monthly': 27.9,
  'optimal_pro|yearly': 249 / 12,
}

type AbonnementMinimal = { abonnement_type?: string | null; abonnement_periode?: string | null }

export function calculeMRR(abonnes: AbonnementMinimal[]): number {
  let mrr = 0
  for (const a of abonnes) {
    const type = a.abonnement_type === 'optimal_pro' ? 'optimal_pro' : 'optimal_particulier'
    const periode = a.abonnement_periode === 'yearly' ? 'yearly' : 'monthly'
    mrr += MRR_PAR_ABONNEMENT[`${type}|${periode}`] || 0
  }
  return Math.round(mrr * 100) / 100
}

export function repartitionAbonnes(abonnes: AbonnementMinimal[]) {
  const r = { particulier_mensuel: 0, particulier_annuel: 0, pro_mensuel: 0, pro_annuel: 0 }
  for (const a of abonnes) {
    const pro = a.abonnement_type === 'optimal_pro'
    const annuel = a.abonnement_periode === 'yearly'
    if (pro && annuel) r.pro_annuel++
    else if (pro) r.pro_mensuel++
    else if (annuel) r.particulier_annuel++
    else r.particulier_mensuel++
  }
  return r
}

// Invitation a laisser un avis Google, une seule fois, apres un premier achat.
// Ciblage : clients non abonnes (les abonnes voient l'invitation dans leur espace client).
// La reclamation atomique sur avis_google_propose garantit un envoi unique et evite tout
// doublon avec l'invitation in-app.
async function proposerAvisGoogleApresAchat(userId: string | null): Promise<void> {
  if (!userId || !AVIS_GOOGLE.lien) return
  try {
    const { data: u } = await supabaseAdmin
      .from('users')
      .select('email, prenom, abonnement_actif, avis_google_propose')
      .eq('id', userId)
      .maybeSingle()
    if (!u || !u.email) return
    if (u.abonnement_actif) return
    if (u.avis_google_propose) return
    const { data: claim } = await supabaseAdmin
      .from('users')
      .update({ avis_google_propose: true })
      .eq('id', userId)
      .eq('avis_google_propose', false)
      .select('id')
    if (!claim || claim.length === 0) return
    await envoyerInvitationAvis(u.email, u.prenom || '')
  } catch (e) {
    console.error('[Webhook] erreur proposition avis Google:', e)
  }
}

  let event: Stripe.Event

  try {
    event = stripe.webhooks.constructEvent(
      body,
      signature,
      process.env.STRIPE_WEBHOOK_SECRET!
    )
  } catch (error) {
    console.error('Webhook signature error:', error)
    return NextResponse.json({ error: 'Invalid signature' }, { status: 400 })
  }

  switch (event.type) {

    case 'checkout.session.completed': {
      const session = event.data.object as Stripe.Checkout.Session
      const priceKey = session.metadata?.priceKey
      const customerEmail = session.customer_email
      const metaUserId = session.metadata?.userId || null

      if (!customerEmail) break

      // RÃ©soudre userId : metadata en prioritÃ©, sinon lookup par email
      const resolveUserId = async (): Promise<string | null> => {
        if (metaUserId) return metaUserId
        const { data } = await supabaseAdmin
          .from('users')
          .select('id')
          .eq('email', customerEmail)
          .single()
        return data?.id || null
      }

      // Abonnement Optimal
      if (priceKey?.includes('optimal')) {
        const isYearly = priceKey.includes('yearly')
        const dateExpiration = new Date()
        dateExpiration.setMonth(dateExpiration.getMonth() + (isYearly ? 12 : 1))

        const updateData = {
          abonnement_actif: true,
          abonnement_type: priceKey.includes('pro') ? 'optimal_pro' : 'optimal_particulier',
          abonnement_periode: isYearly ? 'yearly' : 'monthly',
          stripe_customer_id: session.customer as string,
          date_expiration: dateExpiration.toISOString(),
        }

        const userId = await resolveUserId()
        if (userId) {
          await supabaseAdmin.from('users').update(updateData).eq('id', userId)
        } else {
          await supabaseAdmin.from('users').update(updateData).eq('email', customerEmail)
        }
      }

      // Question unitaire ou abonne
      if (priceKey === 'question_unitaire' || priceKey === 'question_abonne') {
        const userId = await resolveUserId()
        await supabaseAdmin
          .from('questions')
          .insert({
            user_id: userId,
            question: null,
            statut: 'en_attente',
            type: priceKey === 'question_unitaire' ? 'unitaire' : 'abonne',
            stripe_payment_id: session.payment_intent as string,
          })
      }

      // Forfait conseil assurances
      if (priceKey?.includes('conseil_assurances')) {
        const userId = await resolveUserId()
        await supabaseAdmin
          .from('dossiers_assurances')
          .insert({
            user_id: userId,
            statut: 'en_attente',
            type: priceKey.includes('pro') ? 'pro' : 'particulier',
            stripe_payment_id: session.payment_intent as string,
          })
      }

      // Achat de document a l'unite (carte) : acces accorde via achats_documents
      if (session.metadata?.produit === 'document' && session.metadata?.documentId) {
        const userId = await resolveUserId()
        const paymentId = session.payment_intent as string
        // Idempotence : ne pas inserer deux fois pour la meme transaction Stripe
        const { data: existant } = await supabaseAdmin
          .from('achats_documents')
          .select('id')
          .eq('stripe_payment_id', paymentId)
          .limit(1)
        if (userId && (!existant || existant.length === 0)) {
          await supabaseAdmin
            .from('achats_documents')
            .insert({
              user_id: userId,
              document_id: session.metadata.documentId,
              stripe_payment_id: paymentId,
              montant: session.amount_total != null ? session.amount_total / 100 : null,
              mode_paiement: 'carte',
              statut: 'paye',
            })
          await proposerAvisGoogleApresAchat(userId)
        }
      }

      // Achat groupe (panier) : acces accorde pour chaque document de la commande
      if (session.metadata?.produit === 'panier' && session.metadata?.documentIds) {
        const userId = await resolveUserId()
        const paymentId = session.payment_intent as string
        const ids = session.metadata.documentIds.split(',').map((s) => s.trim()).filter(Boolean)
        if (userId && ids.length > 0) {
          const { data: docs } = await supabaseAdmin
            .from('documents')
            .select('id, prix_unitaire')
            .in('id', ids)
          const prixMap = new Map((docs || []).map((d) => [d.id, d.prix_unitaire]))
          for (const docId of ids) {
            const { data: existant } = await supabaseAdmin
              .from('achats_documents')
              .select('id')
              .eq('stripe_payment_id', paymentId)
              .eq('document_id', docId)
              .limit(1)
            if (!existant || existant.length === 0) {
              await supabaseAdmin
                .from('achats_documents')
                .insert({
                  user_id: userId,
                  document_id: docId,
                  stripe_payment_id: paymentId,
                  montant: prixMap.get(docId) ?? null,
                  mode_paiement: 'carte',
                  statut: 'paye',
                })
            }
          }
          await proposerAvisGoogleApresAchat(userId)
        }
      }

      // Email de confirmation de commande (carte) aux couleurs du cabinet
      {
        let typeCommande: TypeCommande | null = null
        if (priceKey?.includes('optimal')) typeCommande = 'abonnement'
        else if (session.metadata?.produit === 'document') typeCommande = 'document'
        else if (session.metadata?.produit === 'panier') typeCommande = 'panier'
        else if (priceKey === 'question_unitaire' || priceKey === 'question_abonne') typeCommande = 'question'
        else if (priceKey?.includes('conseil_assurances')) typeCommande = 'conseil'

        if (typeCommande) {
          const { data: u } = await supabaseAdmin.from('users').select('prenom').eq('email', customerEmail).maybeSingle()
          await envoyerConfirmationCommande(customerEmail, u?.prenom || '', typeCommande, eur(session.amount_total))
        }
      }
      break
    }

    case 'customer.subscription.updated': {
      const subscription = event.data.object as Stripe.Subscription
      const customerId = subscription.customer as string

      // Resiliation programmee : Stripe positionne cancel_at_period_end (et cancel_at)
      // quand une fin de periode est demandee, et les remet a false/null en cas de reactivation.
      if (subscription.cancel_at_period_end) {
        const finUnix = subscription.cancel_at
        await supabaseAdmin
          .from('users')
          .update({
            resiliation_programmee: true,
            resiliation_fin: finUnix ? new Date(finUnix * 1000).toISOString() : null,
          })
          .eq('stripe_customer_id', customerId)
      } else {
        await supabaseAdmin
          .from('users')
          .update({
            resiliation_programmee: false,
            resiliation_fin: null,
          })
          .eq('stripe_customer_id', customerId)
      }

      break
    }

    case 'customer.subscription.deleted': {
      const subscription = event.data.object as Stripe.Subscription
      const customerId = subscription.customer as string

      await supabaseAdmin
        .from('users')
        .update({
          abonnement_actif: false,
          abonnement_type: null,
          date_expiration: null,
          resiliation_programmee: false,
          resiliation_fin: null,
        })
        .eq('stripe_customer_id', customerId)

      break
    }

    case 'invoice.payment_failed': {
      const invoice = event.data.object as Stripe.Invoice
      const customerId = invoice.customer as string

      await supabaseAdmin
        .from('users')
        .update({ abonnement_actif: false })
        .eq('stripe_customer_id', customerId)

      break
    }
  }

  return NextResponse.json({ received: true })
}

import { createClient } from '@supabase/supabase-js'
import { NextResponse } from 'next/server'

export async function GET(request: Request) {
  const { searchParams, origin } = new URL(request.url)
  const code = searchParams.get('code')
  const type = searchParams.get('type')

  if (code) {
    const supabase = createClient(
      process.env.NEXT_PUBLIC_SUPABASE_URL!,
      process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
    )
    await supabase.auth.exchangeCodeForSession(code)
  }

  // Flux recovery : rediriger vers la page de saisie du nouveau mot de passe
  if (type === 'recovery') {
    return NextResponse.redirect(`${origin}/nouveau-mot-de-passe`)
  }

  // Flux standard (inscription, connexion) : espace client
  return NextResponse.redirect(`${origin}/espace-client`)
}

  try {

    return readdirSync(blogDir, { withFileTypes: true })
      .filter(

          !e.name.startsWith('[') &&
          !e.name.startsWith('.') &&
          !e.name.startsWith('_'),

      .filter((e) => existsSync(join(blogDir, e.name, 'page.tsx')))
      .map((e) => ({
        url: `${baseUrl}/blog/${e.name}`,
        lastModified,
        changeFrequency: 'monthly' as Freq,
        priority: 0.6,
      }))
  } catch {
    return []
  }

    if (!data) return []
    const now = new Date()
    const vus = new Set<string>()
    const urls: MetadataRoute.Sitemap = []
    for (const c of data) {
      const nom = c.nom ? String(c.nom).trim() : ''
      if (!nom) continue
      const s = slugify(nom)
      if (!s || vus.has(s)) continue
      vus.add(s)
      urls.push({ url: `${baseUrl}/catalogue/categorie/${s}`, lastModified: now, changeFrequency: 'weekly' as Freq, priority: 0.6 })
    }
    return urls
  } catch {
    return []
  }
}

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const lastModified = new Date()
  const statiques: MetadataRoute.Sitemap = pages.map((p) => ({
    url: p.path === '/' ? baseUrl : `${baseUrl}${p.path}`,
    lastModified,
    changeFrequency: p.changeFrequency,
    priority: p.priority,
  }))
  const articles = getArticlesBlog()
  const ressources = await getRessources()
  const categories = await getCategories()
  return [...statiques, ...articles, ...ressources, ...categories]
}
